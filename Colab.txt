server_code = """
import os

app = Flask(__name__)

class Config:

        self.extractor_path = 'networks/extractor.pth'
        self.upscaler_path = 'networks/RealESRGAN_x4plus_anime_6B.pt'

        self.colorizer_type = 'AlacGAN'
        self.upscaler_type = 'ESRGAN'

        # Tile size 0 means "Process whole image" (Faster, but uses more VRAM)
        # Since we have 16GB VRAM and small models, we can afford this.
        self.alacgan = ModelSettings()
        self.alacgan.tile_size = 0
        self.alacgan.tile_pad = 0
        self.alacgan.image_size = 576

        self.esrgan = ModelSettings()
        self.esrgan.tile_size = 256

        self.gigagan = ModelSettings()

config = Config()

# --- WORKER POOL SETUP (MAX PERFORMANCE) ---
# Previous: 2 Workers (~2GB VRAM)
# New: 6 Workers (~7-8GB VRAM) -> This uses half the GPU, leaving room for Upscaling spikes.
NUM_WORKERS = 6
model_queue = queue.Queue()

try:
    print(f"[-] Initializing {NUM_WORKERS} High-Speed Workers on {config.device}...", flush=True)

    for i in range(NUM_WORKERS):
        print(f"    ... Loading Worker {i+1}/{NUM_WORKERS}", flush=True)
        # Create fresh instances
        denoiser = MangaDenoiser(config)
        colorizer = MangaColorizator(config)
        upscaler = MangaUpscaler(config)

        # Warmup (Optional: Runs a dummy black image through to allocate memory)
        # dummy = np.zeros((100, 100, 3), dtype=np.uint8)
        # denoiser.denoise(dummy, sigma=25)

        model_queue.put({
            'id': i+1,
            'denoiser': denoiser,
            'colorizer': colorizer,
            'upscaler': upscaler
        })

    print(f"[+] {NUM_WORKERS} Workers Ready! System is running at Max Speed.", flush=True)

except Exception as e:
    print(f"[!] Critical Error Loading Models: {e}", flush=True)
    sys.exit(1)

def fast_color_check(pil_image, tolerance=32):
    try:
        thumb = pil_image.resize((128, 128), resample=PIL.Image.BILINEAR)
        img_np = np.array(thumb)
        img_bgr = cv2.cvtColor(img_np, cv2.COLOR_RGB2BGR)
        hsv = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2HSV)
        s_channel = hsv[:, :, 1]
        mean_sat = np.mean(s_channel)
        high_sat_pixels = np.count_nonzero(s_channel > tolerance)
        pixel_ratio = high_sat_pixels / s_channel.size
        return (mean_sat > tolerance) or (pixel_ratio > 0.15), mean_sat
    except:
        return False, 0

def adjust_saturation(img_np, factor):
    if factor == 1.0: return img_np
    pil_img = PIL.Image.fromarray(img_np)
    enhancer = PIL.ImageEnhance.Color(pil_img)
    pil_img = enhancer.enhance(factor)
    return np.array(pil_img)

@app.route('/', methods=['GET', 'POST'])
def proxy():
    target_url = request.args.get('url')
    if not target_url: return f"Proxy Active ({NUM_WORKERS} Workers)", 200

    if '127.0.0.1' in target_url or 'localhost' in target_url:
        return "Localhost Skipped", 400

    print(f"[*] Fetching: {target_url}", flush=True)

    referer_candidates = [
        'https://comick.live/', 'https://comick.io/', 'https://comick.app/', None
    ]

    base_headers = {
        'Accept': 'image/avif,image/webp,image/apng,image/svg+xml,image/*,*/*;q=0.8',
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36',
    }

    tachi_cookies = None
    headers_str = request.args.get('headers')
    if headers_str:
        try:
            tachi_headers = json.loads(headers_str)
            if 'Cookie' in tachi_headers: tachi_cookies = tachi_headers['Cookie']
        except: pass

    r = None
    success_download = False

    for ref in referer_candidates:
        current_headers = base_headers.copy()
        if ref: current_headers['Referer'] = ref
        if tachi_cookies: current_headers['Cookie'] = tachi_cookies

        try:
            r = crequests.get(target_url, headers=current_headers, impersonate="chrome124", timeout=15, allow_redirects=True)
            if r.status_code == 200:
                success_download = True
                break
            elif r.status_code == 403: continue
            else: break
        except: break

    if not success_download or r is None:
        if r: return Response(r.content, status=r.status_code)
        return "Download Failed", 500

    try:
        image_bytes = BytesIO(r.content)
        try:
            pil_img = PIL.Image.open(image_bytes).convert("RGB")
        except:
            return Response(r.content, status=200)

        is_colored, score = fast_color_check(pil_img, config.color_tolerance)
        if is_colored:
            print(f"    -> [SKIP] Already colored (Sat: {score:.1f})", flush=True)
            image_bytes.seek(0)
            return send_file(image_bytes, mimetype='image/jpeg')

        # --- HIGH SPEED PARALLEL PROCESSING ---
        worker = model_queue.get()
        try:
            # print(f"    -> Assigned to Worker {worker['id']}", flush=True)
            img_np = np.array(pil_img)

            img_np = worker['denoiser'].denoise(img_np, sigma=config.denoise_sigma)
            worker['colorizer'].set_image((img_np.astype('float32') / 255), config.colorized_image_size)
            img_np = worker['colorizer'].colorize()
            img_np = worker['upscaler'].upscale((img_np.astype('float32') / 255), 4)

            # Manual Garbage Collection helps when running 6+ workers to prevent fragmentation
            # torch.cuda.empty_cache() # Uncomment if you hit OOM, but slows it down slightly

        finally:
            model_queue.task_done()
            model_queue.put(worker)
        # --------------------------------------

        img_np = adjust_saturation(img_np, config.saturation_factor)

        img_bgr = cv2.cvtColor(img_np, cv2.COLOR_RGB2BGR)
        success, encoded_img = cv2.imencode('.webp', img_bgr, [int(cv2.IMWRITE_WEBP_QUALITY), 85])

        if success:
            return send_file(BytesIO(encoded_img.tobytes()), mimetype='image/webp')
        else:
            return Response(r.content, status=200)

    except Exception as e:
        print(f"[!] Processing Error: {e}", flush=True)
        return Response(r.content, status=200)

if __name__ == '__main__':
    print("READY_TO_SERVE", flush=True)
    app.run(host='0.0.0.0', port=5000, threaded=True)

flask_ready = False
def monitor_flask():
    global flask_ready
    while True:
        line = flask_process.stdout.readline()
        if not line: break
        line = line.strip()
        print(f"[FLASK] {line}")
        if "READY_TO_SERVE" in line:
            flask_ready = True

t = threading.Thread(target=monitor_flask, daemon=True)
